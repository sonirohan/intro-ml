\section{Setup, Generalization \& Data Splits}
Regression: $y\in\R$; classification: $y\in\{1..c\}$. Learn $f$ s.t.\ $\hat y=f(\x)\approx y$ on \emph{unseen} data (same distribution).
\textbf{Overfit}: train err low, val/test high (high variance). \textbf{Underfit}: train err high (high bias).
\textbf{Train} = fit params. \textbf{Val} = pick hyperparams. \textbf{Test} = once, at the end. Tuning on test $\Rightarrow$ optimistic (overfit to test).
\textbf{$k$-fold CV}: $k$ models fitted; CV score = \emph{mean} val perf over folds; then retrain on all, test once.
\textbf{Leakage}: highly correlated samples (e.g.\ subsampled time series) split across train/val $\Rightarrow$ val acc inflated. \emph{Split first, then subsample/augment.}
Estimator: bias $=\E[\hat z]-z$, var $=\E[(\hat z-\E\hat z)^2]$.
\textbf{Params} (learned by GD): $\W,\bb$, BN $\gamma,\beta$, PReLU $\alpha$. \textbf{Hyperparams}: LR, batch size, $k$, $\lambda$, dropout $p$, loss fn, \#layers, $\beta_1,\beta_2$.

\subsection{Diagnosis (very common short answer)}
\begin{itemize}
\item Overfit fix: more data, data aug., \emph{reduce} complexity, L1/L2, dropout, early stop, BN, ensembles, label smoothing. NOT more layers.
\item Underfit fix: bigger model, less reg., train longer. Adding reg./dropout/less data does \emph{not} help underfit.
\item Loss oscillates/diverges: lower LR or adaptive (Adam/RMSProp). Loss flat from start w/ tiny LR ($10^{-6}$): raise LR.
\item Sudden step drop in loss curve: LR decay (annealing).
\item Smooth monotone curve = full-batch GD; noisy = minibatch/SGD (noisy grad est.).
\item Reg.\ never aims to lower \emph{train} loss; too much $\Rightarrow$ underfit (can hurt test too).
\item Optimizing accuracy directly: not differentiable (zero grad a.e.) $\Rightarrow$ use CE proxy. Lower CE can coexist w/ a misclassification.
\end{itemize}

\section{$k$-NN \& Linear Classifiers}
\textbf{$k$-NN}: majority vote of $k$ nearest (metric = hyperparam). Train = store. $k{=}1$ \& pt is own neighbor $\Rightarrow$ train err 0. $\uparrow k$: smoother boundary, robust to noise/outliers, \emph{less} overfit. Pick $k$ by CV (never by train err). Cons: memory, slow test $O(N)$, curse of dim.\ ($L_2$ in pixel space $\ne$ similarity). Low bias / high variance; linear clf: high bias / low variance.
\textbf{Linear}: $\z=\W\x+\bb$, $\W\in\R^{c\times n}$, rows = class templates; predict $\arg\max_j z_j$. 2-class boundary $(\w_1-\w_2)^T\x=0$.
\textbf{Hinge/SVM}: $\sum_{j\ne y}\max(0,1+a_j-a_y)$; margin 1 WLOG (rescale $\W,\bb$).

\section{Least Squares \& Matrix Calculus}
$\mathcal L=\tfrac12\|\y-\X\theta\|^2=\tfrac12(\y^T\y-2\y^T\X\theta+\theta^T\X^T\X\theta)$
\[\nabla_\theta\mathcal L=\X^T\X\theta-\X^T\y\ \Rightarrow\ \theta^*=(\X^T\X)^{-1}\X^T\y\]
GD: $\theta\leftarrow\theta-\epsilon(\X^T\X\theta-\X^T\y)$. Ridge ($+\frac\lambda2\|\theta\|^2$): $\theta\leftarrow(1-\epsilon\lambda)\theta-\epsilon\nabla\mathcal L$; closed form $(\X^T\X+\lambda\mathbf I)^{-1}\X^T\y$.

\textbf{Denominator layout}: $\nabla_\x y$ same shape as $\x$; $\nabla_\x\y\in\R^{m\times n}=\mathbf J^T$.
\begin{itemize}
\item $\nabla_\x(\mathbf a^T\x)=\mathbf a$;\ $\nabla_\x(\x^T\A\x)=(\A+\A^T)\x$;\ $\nabla_\x(\W\x)=\W^T$
\item $\nabla_\A(\z^T\A\x)=\z\x^T$;\ $\nabla_\W\|\W\|_F^2=2\W$;\ $\nabla_\X\log|\X|=\X^{-T}$
\item $\nabla_\x\|\x\|_2=\x/\|\x\|_2$;\ quotient: $(f/g)'=(f'g-fg')/g^2$
\end{itemize}
\textbf{Chain rule}: $\x\to\y\to\z$: $\nabla_\x\z=\nabla_\x\y\,\nabla_\y\z$ \key{(right-to-left; check dims)}. Hessian $H_{ij}=\partial^2f/\partial x_i\partial x_j$.
Geometric: $\sum_{j=0}^{n-1}r^j=\frac{1-r^n}{1-r}$.

\section{MLE \& Probabilistic Losses}
$\theta^*=\arg\max\prod_i p(y^{(i)}|\x^{(i)},\theta)=\arg\min-\sum_i\log p(\cdot)$ ($p(\x)$ indep.\ of $\theta$; log monotone).
\textbf{Bernoulli/coin}: $\theta^*=\frac{n_H}{n_H+n_T}$. \textbf{Poisson} $\frac{\lambda^ke^{-\lambda}}{k!}$: $\ell=-n\lambda+\ln\lambda\sum x_j-\sum\ln x_j!$, $\hat\lambda=\bar x$. \textbf{Gaussian} noise $\Rightarrow$ MSE.
\textbf{BCE = MLE}: $p(y|\x)=\hat y^{y}(1-\hat y)^{1-y}$ $\Rightarrow$ $-\log L=-\sum[y\log\hat y+(1-y)\log(1-\hat y)]$.
\textbf{LDA$\to$softmax}: $p(\x|c)=\mathcal N(\mu_c,\Sigma)$, prior $\pi_c$: $p(c|\x)=$ softmax with $\w_c=\Sigma^{-1}\mu_c$, $b_c=-\frac12\mu_c^T\Sigma^{-1}\mu_c+\log\pi_c$ (the $\x^T\Sigma^{-1}\x$ term cancels).

\section{Softmax Classifier}
$a_j=\w_j^T\x+b_j$, $p_j=\frac{e^{a_j}}{\sum_k e^{a_k}}$. $\mathcal L_i=\log\sum_je^{a_j}-a_{y_i}=-\log p_{y_i}$.
\[\pd{\mathcal L_i}{a_k}=p_k-\bI[k{=}y_i]\quad(\text{vector: }\mathbf p-\mathbf e_{y_i})\]
$\pd{\mathcal L}{\w_k}=\sum_i(p_k^{(i)}-\bI[k{=}y_i])\x^{(i)}$, $\pd{\mathcal L}{b_k}=\sum_i(p^{(i)}_k-\bI[k{=}y_i])$; matrix: $\nabla_\W=(\mathbf p-\mathbf e_y)\x^T$. \ex loss at init (small rand.\ $\W$) $\approx\ln C$ ($C{=}10$: 2.30; $C{=}7$: 1.95). CE of one example $=-\log\hat y_{\text{true}}$ (e.g.\ $-\log0.3$).
Overflow: subtract $\max_j a_j$ from all scores. Label smoothing: targets $1-\epsilon$ and $\frac{\epsilon}{k-1}$ (regularizer; robust to label errors).

\section{Neural Nets}
$\h_1=f(\W_1\x+\bb_1),\dots,\z=\W_N\h_{N-1}+\bb_N$ (no act.\ on output). \textbf{Params}: $\sum(n_{in}n_{out}+n_{out})$; e.g.\ $3{\to}4{\to}2$: 26. Residual $\h_1=f(\W_1\x+\bb_1)+\W_s\x$: $2mn+m+ml+l$.
Linear $f$ collapses to one linear map ($\W=\W_N\cdots\W_1$, $\bb=\bb_N+\W_N\bb_{N-1}+\dots$); only useful for low-rank (autoencoder $\approx$ SVD). Nonlinear hidden layers learn features that make data linearly separable. Universal approx: 1 hidden layer, enough units.
\textbf{XOR}: linear LS fails; $g(\x)=\w^T\max(0,\W^T\x+\mathbf c)$, $\W=\begin{bmatrix}1&1\\1&1\end{bmatrix}$, $\mathbf c=[0,-1]^T$, $\w=[1,-2]^T$ works.

\section{Activation Functions}
\resizebox{\columnwidth}{!}{\begin{tabular}{@{}l@{\ }l@{\ }c@{\,}c@{\,}c@{\,}c@{}}
 & $f'(x)$ & Sat. & ZZ & Diff & Id$_0$\\\hline
$\sigma=\frac1{1+e^{-x}}$ & $\sigma(1-\sigma)\le\frac14$ & $\pm\infty$ & Y & Y & N\\
$\tanh=2\sigma(2x){-}1$ & $1-\tanh^2$ & $\pm\infty$ & N & Y & Y\\
ReLU $\max(0,x)$ & $\bI(x>0)$ & $x<0$ & Y & N & N\\
Leaky/PReLU $\max(\alpha x,x)$ & $1$ / $\alpha$ & no & N & N & N\\
ELU $x$ / $\alpha(e^x{-}1)$ & $1$ / $\alpha e^x$ & $-\infty$ & N & N$^*$ & N\\
Softplus $\log(1{+}e^x)$ & $\sigma(x)=\frac{e^x}{1+e^x}$ & $-\infty$ & Y & Y & N\\
$e^x$ & $e^x$ & $-\infty$ & Y & Y & N\\
GELU $x\Phi(x)$ & $\Phi{+}x\phi$ ($\frac12$ at 0) & $-\infty$ & N & Y & N\\
\end{tabular}}\par
ZZ = zig-zag (outputs all $\ge0$ $\Rightarrow$ $\partial f/\partial\w=f'\cdot\h$ entries share sign). Id$_0$: $f(0)=0,f'(0)=1$. $^*$ELU diff.\ iff $\alpha=1$. PReLU: $\partial f/\partial\alpha=x\,\bI(x<0)$; learnable $\alpha$ avoids dead units \& zig-zag.
\ex Vanishing grads: sigmoid, tanh (saturation), ReLU ($x<0$). Step/sign fn: $f'=0$ a.e.\ $\Rightarrow$ no learning. Sigmoid outputs $\sim0.5$ (not 0-centered). ReLU: fast, no sat.\ for $x>0$, dead units.
\textbf{Output$\leftrightarrow$loss}: $\sigma$+MSE: $\partial/\partial z=-(y-\sigma)\sigma(1-\sigma)\to0$ when very wrong. $\sigma$+CE: $\partial/\partial z=\sigma(z)-y$ (learns). Pair CE with sigmoid/softmax.

\section{Backprop}
Forward: compute \& cache node values. Backward: downstream = local $\times$ upstream. \textbf{add} distributes; \textbf{mult} swaps ($\partial_x=y\cdot$up); \textbf{max} routes to the winner. E.g.\ $f=(x+y)z$, $(2,3,4)$: $\partial_z=5$, $\partial_x=\partial_y=4$. Sigmoid gate: local $=\sigma(1-\sigma)$ (e.g.\ $0.73\cdot0.27=0.20$). Write intermediate vars, label graph edges, give dims. \key{Paths merge $\Rightarrow$ sum} (shared weights, siamese nets, residual, $\x$ used twice).
\textbf{Elementwise} $\h=f(\mathbf a)$: $\delta_{\mathbf a}=f'(\mathbf a)\odot\delta_\h$.
\textbf{Affine} $\mathbf c=\W\x+\bb$: $\delta_\x=\W^T\delta_{\mathbf c}$, $\delta_\W=\delta_{\mathbf c}\x^T$, $\delta_\bb=\delta_{\mathbf c}$. Row-vector $\W_2\in\R^{1\times d}$: $\delta_{\W_2}=\delta_z\,\mathbf a_1^T$.
\textbf{Template} (MLP, $\hat y=\sigma(z_2)$): $\delta_{z_2}=\hat y(1-\hat y)\delta_{\hat y}$ [$=\hat y-y$ for BCE]; $\delta_{\W_2}=\delta_{z_2}\mathbf a_1^T$, $\delta_{\bb_2}=\delta_{z_2}$; $\delta_{\mathbf a_1}=\W_2^T\delta_{z_2}$; $\delta_{\z_1}=\bI(\z_1>0)\odot\delta_{\mathbf a_1}$; $\delta_{\W_1}=\delta_{\z_1}\x^T$, $\delta_{\bb_1}=\delta_{\z_1}$.
\textbf{Siamese} ($\h_p=\W_1\x_p+\bb_1$, $\h_q=\W_1\x_q+\bb_1$): $\delta_{\W_1}=\delta_{\h_p}\x_p^T+\delta_{\h_q}\x_q^T$, $\delta_{\bb_1}=\delta_{\h_p}+\delta_{\h_q}$. Residual $\z=\W_3f(\W_1\x)+\W_2\x$: $\delta_\x=\W_1^T[f'\odot\W_3^T\delta_\z]+\W_2^T\delta_\z$.
Nullspace $\W_1\mathbf z=0$: replacing $\x\to\x+\mathbf z$ changes only $\delta_{\W_1}=\delta_\h(\x+\mathbf z)^T$.

\subsection{Upstream (loss) gradients}
\begin{itemize}
\item $\frac12\|\z-\y\|^2$: $\z-\y$.\quad $\|\h-\z\|^2$: $2(\h-\z)$
\item BCE wrt $\hat y$: $-\frac{y}{\hat y}+\frac{1-y}{1-\hat y}$; weighted $-[\alpha y\log\hat y+\beta(1-y)\log(1-\hat y)]$: $-\frac{\alpha y}{\hat y}+\frac{\beta(1-y)}{1-\hat y}$. Balance: weight $\propto$ 1/class size (10k neg : 1k pos $\Rightarrow\alpha{=}10,\beta{=}1$).
\item Hinge $\sum_i\max(0,1-y_i\hat y_i)$: $-y_i$ where active ($\hat y\in(0,1)$, $y\in\{0,1\}\Rightarrow$ always active).
\item Cosine $s=\frac{\z^T\hat\z}{\|\z\|\|\hat\z\|}$: $\pd s\z=\frac{\hat\z}{\|\z\|\|\hat\z\|}-\frac{s\,\z}{\|\z\|^2}$. $s{=}1$ same dir., 0 orthogonal, $-1$ opposite.
\item $(\log|\h\h^T+\mathbf I|-c)^2$: with $\mathbf M=\h\h^T+\mathbf I$: $\delta_\h=4(\cdot)\mathbf M^{-T}\h$.
\end{itemize}

\subsection{Penalty gradients (add to $\nabla$ of data loss)}
\begin{itemize}
\item $\frac\lambda2\|\w\|_2^2\to\lambda\w$;\ $\lambda\|\W\|_F^2\to2\lambda\W$;\ $\lambda\|\w\|_1\to\lambda\sgn(\w)$
\item $\|\A\|_\infty=\max_k\sum_j|A_{kj}|$ (max abs row sum): $\nabla=$ zeros except row $j^*$ (the max row) $=\sgn(\A_{j^*,:})$
\item $\text{LSE}(\x)=\ln\sum_ie^{x_i}\to\text{softmax}(\x)$
\item $\|\w_1-\w_k\|^2$ (tie layers): $\pm2(\w_1-\w_k)$. $\|\h\|_1$: sparse activations. $\|\h-\mathbf 1\|^2$: acts $\to1$.
\item Elastic net $\frac1{2n}\|\y-\X\theta\|^2+\lambda[\alpha\|\theta\|_1+\frac{1-\alpha}2\|\theta\|^2]$: $\nabla=\frac1n\X^T(\X\theta-\y)+\lambda[\alpha\sgn\theta+(1-\alpha)\theta]$. $\alpha{=}1$: L1 (sparse); $\alpha{=}0$: L2 (shrink). Combines sparsity with stability for correlated features.
\end{itemize}

\section{Weight Initialization}
\textbf{All zero} (no bias, tanh): zero grads, no learning. \textbf{All same const}: symmetry not broken; all units in a layer get identical grads (stay equal). \textbf{Small} (deep): acts shrink each layer $\to0$ $\Rightarrow$ $\delta_\W=\delta\h^T\approx0$. \textbf{Large}: tanh saturates (vanishing), ReLU explodes.
$\text{var}(wh)=\E^2[w]\text{var}(h)+\E^2[h]\text{var}(w)+\text{var}(w)\text{var}(h)$ $\Rightarrow$ $\text{var}(h_i)=n_{in}\text{var}(w)\text{var}(h_{i-1})$ (zero-mean, indep.); fwd wants $\text{var}(w)=\frac1{n_{in}}$, bwd $\frac1{n_{out}}$. \textbf{Xavier}: $\text{var}(w)=\frac{2}{n_{in}+n_{out}}$ (tanh). \textbf{He}: $\frac2{n_{in}}$ (ReLU kills half).

\section{Batch / Layer Normalization}
Per unit over minibatch: $\hat x=\frac{x-\mu_B}{\sqrt{\sigma_B^2+\varepsilon}}$, $y=\gamma\hat x+\beta$ ($\gamma,\beta$ \emph{learned}; can undo BN). Placed before nonlinearity. Test: running avg.\ of $\mu,\sigma^2$.
\ex BN is \emph{affine/linear} per unit, learnable, adds noise (batch estimates) $\Rightarrow$ mild regularizer, speeds training (higher LR), robust to init (acts can't vanish/explode; reduces internal covariate shift).
\textbf{Backprop}: $\partial_\beta=\sum_j\delta_{y_j}$, $\partial_\gamma=\sum_j\delta_{y_j}\hat x_j$, $\delta_{\hat x_j}=\gamma\delta_{y_j}$;
$\delta_{\sigma^2}=\sum_j\delta_{\hat x_j}(x_j-\mu)\cdot(-\tfrac12)(\sigma^2+\varepsilon)^{-3/2}$;
$\delta_\mu=-\frac{\sum_j\delta_{\hat x_j}}{\sqrt{\sigma^2+\varepsilon}}-\frac2m\sum_j(x_j-\mu)\,\delta_{\sigma^2}$;
$\delta_{x_j}=\frac{\delta_{\hat x_j}}{\sqrt{\sigma^2+\varepsilon}}+\frac{2(x_j-\mu)}m\delta_{\sigma^2}+\frac1m\delta_\mu$ (3 paths).
\textbf{LayerNorm}: stats over the $D$ units of \emph{one example} (not over batch); same at train/test; better for small batches, RNNs/sequences, transformers.

\section{Regularization}
Reduce generalization (not train) error: $\uparrow$bias, $\downarrow$variance. Early stop, norm penalties, aug., dropout, ensembles, multitask, BN, noise, label smoothing are all regularizers.
\textbf{L2 = weight decay}: $\w\leftarrow(1-\epsilon\lambda)\w-\epsilon\nabla J$ (shrink by const.\ factor each step). Quadratic approx.: $\tilde w_i=\frac{h_i}{h_i+\lambda}w_i^*$ (shrinks low-curvature dirs most).
\textbf{L1 $\Rightarrow$ sparsity}: with $J\approx J(\w^*)+\frac12(\w-\w^*)^T\Hm(\w-\w^*)$, $\Hm=\text{diag}(h_i)$: $\nabla_{w_i}=h_i(w_i-w_i^*)+\lambda\sgn(w_i)=0$ $\Rightarrow$ $w_i=\sgn(w_i^*)\max\big(|w_i^*|-\frac{\lambda}{h_i},0\big)$ (exactly 0 if $|w_i^*|\le\lambda/h_i$). Use L1 to ``shrink effective model size'' (L2/dropout don't zero weights).
\textbf{Early stopping}: plot loss vs.\ iterations; train loss keeps $\downarrow$; val loss U-shaped; stop at val minimum.
\textbf{Data aug.}: crops, shifts, rotations, scaling, color, blur, flips \emph{only if label-invariant} (no flips for digits). Helps class imbalance (augment minority).
\textbf{Ensembles}: models make partly independent errors $\Rightarrow$ averaging wrong only when many err together. Bagging: resample w/ replacement.
\textbf{Multitask}: shared trunk + heads. \textbf{Transfer}: freeze most of pretrained net, train last layers. \textbf{PEFT}: LoRA ($\W_0+\mathbf B\mathbf A$, low rank per layer), prefix tuning, small hypernetwork; backbone frozen.

\section{Dropout}
Train: keep unit w.p.\ $p$: \texttt{M=(rand<p)}. Test: all units, scale acts by $p$. \textbf{Inverted}: \texttt{M=(rand<p)/p} in training so $\E[\text{act}]$ matches test; test unchanged (no extra cost). Why it works: $\approx$ bagging $2^N$ weight-sharing subnets; no co-adaptation/over-reliance on few inputs; redundant features. Dropout at val/test time: never.

\section{SGD, Momentum, Nesterov}
GD $\theta\leftarrow\theta-\epsilon\g$ ($-\g$ = steepest descent from 1st-order Taylor). Too large $\epsilon$: bounce/diverge; too small: slow. On quadratic: $J(\theta-\epsilon\g)\approx J-\epsilon\g^T\g+\frac{\epsilon^2}2\g^T\Hm\g$ (big curvature along $\g$ $\Rightarrow$ step can \emph{increase} $J$). Minibatch $\g=\frac1m\sum_{i\in B}\nabla\ell_i$: unbiased, noisy (not exact grad); noise helps escape. Full batch = exact.
\textbf{Momentum} ($\vv{=}0$, $\alpha{=}0.9$): $\boxed{\vv\leftarrow\alpha\vv-\epsilon\g,\ \theta\leftarrow\theta+\vv}$. Keeps extra state $\vv$ (running avg.\ of grads). Averages out zig-zag; carries through flat regions; at a min $\g\approx0$ but $\vv\ne0$ $\Rightarrow$ can leave sharp local minima, prefers wide/flat minima (generalize better).
\textbf{Nesterov}: $\boxed{\vv\leftarrow\alpha\vv-\epsilon\nabla J(\theta+\alpha\vv),\ \theta\leftarrow\theta+\vv}$ (look ahead, then correct). $\nabla J(\theta+\alpha\vv)\approx\nabla J(\theta)+\alpha\Hm\vv$: extra $-\epsilon\alpha\Hm\vv$ opposes the update in high curvature $\Rightarrow$ less overshoot.
\textbf{Reparam.} $\tilde\theta=\theta+\alpha\vv$: $\vv_{new}=\alpha\vv_{old}-\epsilon\nabla J(\tilde\theta_{old})$, $\tilde\theta_{new}=\tilde\theta_{old}+\vv_{new}+\alpha(\vv_{new}-\vv_{old})$.
\emph{Proof}: $\tilde\theta_{new}=\theta_{new}+\alpha\vv_{new}=\theta_{old}+\vv_{new}+\alpha\vv_{new}=\tilde\theta_{old}-\alpha\vv_{old}+(1+\alpha)\vv_{new}$.
Code: \texttt{p\_old=p; p=a*p-eps*g; x+=p+a*(p-p\_old)}.

\subsection{Unrolled forms ($\vv_0=0$)}
$\vv_t=-\epsilon\sum_{i=1}^t\alpha^{t-i}\g_i$;\quad $\theta_t=\theta_0+\sum_{j=1}^t\vv_j=\theta_0-\epsilon\sum_{j=1}^t\sum_{i=1}^j\alpha^{j-i}\g_i$
$=\theta_0-\epsilon\sum_{i=1}^t\frac{1-\alpha^{t-i+1}}{1-\alpha}\g_i$. If $\E\g_i=\mu$, $\E\theta_0=0$:
\[\E\theta_t=-\tfrac{\epsilon\mu}{1-\alpha}\Big[t-\tfrac{\alpha(1-\alpha^t)}{1-\alpha}\Big]\]
If $\E\g_i\g_j^T=\sigma^2\mathbf I\,\delta_{ij}$, zero mean: $\text{Cov}(\vv_t)=\epsilon^2\sigma^2\frac{1-\alpha^{2t}}{1-\alpha^2}\mathbf I$ (cross terms vanish).
Nesterov (reparam form): $\theta_t=\theta_0+\alpha\vv_t+\sum_{j=1}^t\vv_j$.

\section{Adaptive LR (per-parameter)}
Annealing: $\epsilon\leftarrow0.1\epsilon$, \emph{scheduled} (every $N$ iters/epochs) or \emph{event-based} (loss plateaus).
\textbf{Adagrad}: $\av\leftarrow\av+\g\odot\g$, $\theta\leftarrow\theta-\frac{\epsilon}{\sqrt{\av}+\nu}\odot\g$ ($\nu{=}10^{-7}$). Large past grads in a coord.\ $\Rightarrow$ smaller step there; boosts flat directions. \key{Con}: $\av$ only grows $\Rightarrow$ eff.\ LR monotonically $\downarrow$ (can stall early). Const.\ $g$: step $=\epsilon/\sqrt t$.
\textbf{RMSProp}: $\av\leftarrow\beta\av+(1-\beta)\g\odot\g$ ($\beta{=}0.99$), same step. EMA forgets old grads $\Rightarrow$ $\av$ can shrink or grow (fixes Adagrad). Const.\ $g$: step $\to\epsilon\,\sgn(g)$.
\emph{Unroll}: $\av_t=(1-\beta)\sum_{i=1}^t\beta^{t-i}\g_i^2$; e.g.\ $\av_1=(1-\beta)\g_1^2$, $\av_2=\beta(1-\beta)\g_1^2+(1-\beta)\g_2^2$, $\theta_2=\theta_0-\frac{\epsilon\g_1}{\sqrt{\av_1}+\nu}-\frac{\epsilon\g_2}{\sqrt{\av_2}+\nu}$.
\textbf{RMSProp+mom.}: $\vv\leftarrow\alpha\vv-\frac{\epsilon}{\sqrt\av+\nu}\odot\g$, $\theta\leftarrow\theta+\vv$.
\textbf{Adam} ($\beta_1{=}0.9,\beta_2{=}0.999$; momentum step + RMSProp step):
\[\vv\leftarrow\beta_1\vv+(1{-}\beta_1)\g,\quad\av\leftarrow\beta_2\av+(1{-}\beta_2)\g\odot\g\]
\[\tilde\vv=\tfrac{\vv}{1-\beta_1^t},\ \ \tilde\av=\tfrac{\av}{1-\beta_2^t},\quad\theta\leftarrow\theta-\tfrac{\epsilon}{\sqrt{\tilde\av}+\nu}\odot\tilde\vv\]
\textbf{Bias corr.}: $\vv_t=(1-\beta_1)\sum_{i=1}^t\beta_1^{t-i}\g_i$ $\Rightarrow$ $\E\vv_t=(1-\beta_1)\frac{1-\beta_1^t}{1-\beta_1}\mu=(1-\beta_1^t)\mu$; same for $\av$ with $s=\E g^2$. Init at 0 biases moments low; correction enlarges early 2nd moment so first steps aren't huge. $t\to\infty$: $\beta^t\to0$, correction $\to1$. First Adam step ($t{=}1$): $\tilde\vv=\g$, $\tilde\av=\g^2$ $\Rightarrow$ $\Delta\theta\approx-\epsilon\,\sgn(\g)$ per coord.\ (scale-free; w/o correction: $\frac{0.1}{\sqrt{0.001}}\approx3.2\times$ bigger). Robust to hyperparams; state $=\vv,\av,t$.
\textbf{AdamW}: L2 inside Adam ($\g=\nabla J+\lambda\theta$) gets divided by $\sqrt{\av}$ $\Rightarrow$ different eff.\ reg.\ per parameter (not intended). Decouple: $\theta\leftarrow\theta-\alpha\,\text{AdamUpdate}(\nabla J)-\alpha\lambda\theta$. (For plain SGD, L2 $\equiv$ weight decay.) Generalizes better; use over Adam.

\section{Optimizer Reasoning (exam favorites)}
\begin{itemize}
\item \textbf{Contour matching} at same point, same $\epsilon$: \emph{GD+mom}: dominated by steep (large-grad) dir., largest step, zig-zags across narrow valley. \emph{Adagrad}: biased toward small-grad (flat) dir., smallest step. \emph{Adam}: in between, not dominated by either axis. \emph{RMSProp}: smooth path along valley. \emph{SGD}: $\perp$ contours, oscillates in steep dir.\ longest. \emph{Momentum} arrow = $-\epsilon\g$ + $\alpha\times$previous step.
\item \textbf{Plateau ``trap''}: mom.\ \& Adam carry accumulated velocity through; update size mom $>$ Adam $>$ Adagrad (Adagrad's $\av$ big from slope).
\item \textbf{Mom.\ vs.\ Nesterov after steep descent}: Nesterov looks ahead at $\theta+\alpha\vv$ (past the min, grad reverses) $\Rightarrow$ brakes, less overshoot, converges faster.
\item \textbf{At a local min} ($\g=0$): mom.\ step $=\alpha\vv$. Nesterov bigger if look-ahead point is past the bump (grad pushes same way), smaller if look-ahead still in the basin (grad opposes).
\item \textbf{Revisit same $\theta$} (full batch, fixed $\epsilon$): GD same step; Adagrad smaller; RMSProp/Adam/mom.\ can't determine (depends on history).
\item SGD+mom.\ \emph{more} likely to leave a local optimum than SGD.
\item Adagrad eff.\ LR is non-increasing: \textbf{True}. RMSProp fixes Adagrad's vanishing LR via EMA.
\end{itemize}

\section{Reading Descent Pictures (why)}
Key idea: \emph{what sets the step direction?}
\begin{itemize}
\item \textbf{GD}: step $\propto-\g$ $\Rightarrow$ $\perp$ to contour lines, length $\propto$ slope. Narrow valley: $\g$ points mostly \emph{across} the valley $\Rightarrow$ zig-zag, slow progress along it; crawls on plateaus/near saddles (small $\g$).
\item \textbf{Momentum}: $\Delta\theta=\alpha\vv_{prev}-\epsilon\g$ (vector sum: last step + new grad). Consistent components (along valley) add up ($\le\frac{1}{1-\alpha}=10\times$); alternating ones (across) cancel $\Rightarrow$ straighter, faster path. Signs: overshoots the min, swings wide on curves, path keeps going after grad changes; can coast into a \emph{different} basin.
\item \textbf{Nesterov}: same shape, tighter turns, less overshoot (grad taken at look-ahead pt corrects early).
\item \textbf{Adagrad/RMSProp}: divide each coord.\ by its RMS grad $\Rightarrow$ every coord.\ moves $\approx\epsilon$ regardless of slope $\Rightarrow$ direction $\approx\sgn(\g)$: \emph{not} $\perp$ contours, heads $\sim45^\circ$, then straight along flat axis (amplifies small-grad coords., damps steep ones) $\Rightarrow$ no zig-zag, escapes saddles/plateaus. Adagrad: steps \emph{shrink} over time; RMSProp keeps speed (reaches min sooner).
\item \textbf{Adam}: RMSProp-style scaling of the \emph{momentum} $\Rightarrow$ smooth straight path, sign-like direction, slight overshoot/curve from momentum.
\item \textbf{Newton}: $\Hm^{-1}$ rescales elongated contours into circles $\Rightarrow$ points straight at the quadratic's min.
\end{itemize}
\textbf{Lecture demos} (2-D loss w/ saddle, local \& global min): start A: all reach global min; mom./Nesterov overshoot then curl back. Start B: GD stalls near saddle; mom.\ \& Nesterov roll into the \emph{same local} min; Adagrad \& RMSProp reach the \emph{global} min (RMSProp faster); Adam slightly slower than RMSProp.
\textbf{Checklist}: $\perp$ contours, steep-axis zig-zag $\to$ GD. Continues prior direction/overshoot $\to$ momentum. Diagonal, equal-sized axis moves, no zig-zag $\to$ adaptive. Steps keep shrinking $\to$ Adagrad.

\section{2nd-Order Methods \& GD Challenges}
1st-order: use $\g$ only, need LR. 2nd-order: use $\Hm=\nabla^2J$ (curvature): flat $\Rightarrow$ big step, high curvature $\Rightarrow$ small step.
\textbf{Newton}: $J\approx J_0+(\theta-\theta_0)^T\g+\frac12(\theta-\theta_0)^T\Hm(\theta-\theta_0)$ $\Rightarrow$ $\boxed{\theta\leftarrow\theta-\Hm^{-1}\g}$; no LR; exact in 1 step for quadratics (1-D: $\theta-f'/f''$). Neg.\ eigenvalues $\Rightarrow$ ascent steps; fix $(\Hm+\alpha\mathbf I)^{-1}$ ($\alpha\to\infty$: GD w/ LR $1/\alpha$). Inverting $\Hm$: $O(N^3)$, impractical.
\textbf{Quasi-Newton (BFGS)}: update $\Hm^{-1}$ recursively, $\mathbf s=\theta_k-\theta_{k-1}$, $\mathbf y=\nabla J_k-\nabla J_{k-1}$: $\Hm_k^{-1}=(\mathbf I-\frac{\mathbf s\mathbf y^T}{\mathbf y^T\mathbf s})\Hm_{k-1}^{-1}(\mathbf I-\frac{\mathbf y\mathbf s^T}{\mathbf y^T\mathbf s})+\frac{\mathbf s\mathbf s^T}{\mathbf y^T\mathbf s}$. \textbf{L-BFGS}: only last $m$ pairs. Needs full/large batches. \textbf{CG}: directions $\g_k^T\Hm\g_{k-1}=0$; ``Hessian-free''.
\textbf{Exploding grads}: cliffs (repeated multiplication, RNNs) $\Rightarrow$ \emph{gradient clipping} (cap $\|\g\|$). \textbf{Vanishing}: $\W^t=\mathbf U\Lambda^t\mathbf U^{-1}$, grad along $\mathbf u_i$ scaled by $\lambda_i^t$ ($|\lambda|<1$ vanish, $>1$ explode); fix via architecture, init, normalization.

\section{Worked Example: Steep vs.\ Flat Axis}
Constant $\g=(4,\,0.1)$ (steep $\theta_1$, flat $\theta_2$), $\epsilon=0.1$, $\alpha=0.9$, $\nu\approx0$. Steps $\Delta\theta$:
\begin{itemize}
\item \textbf{GD}: $-(0.4,\,0.01)$ every step; ratio 40:1, almost purely steep axis.
\item \textbf{Momentum}: $-(0.4,0.01)$, then $-(0.76,0.019)$, \dots$\to\frac{\epsilon\g}{1-\alpha}=-(4,0.1)$. Grows 10$\times$, same direction as GD.
\item \textbf{Adagrad}: $\av_1=(16,0.01)$: $-(0.1,0.1)$ (45$^\circ$!). $\av_2=(32,0.02)$: $-(0.071,0.071)$; step $t$: $\frac{\epsilon}{\sqrt t}$.
\item \textbf{RMSProp} ($\beta{=}0.99$): $\av_1=0.01\g^2$ $\Rightarrow$ $-(1,1)$ (huge first step: no bias correction); $\av_2\approx0.0199\g^2$: $-(0.71,0.71)$; $\to-(0.1,0.1)$.
\item \textbf{Adam}: $t{=}1$: $\tilde\vv=\g,\tilde\av=\g^2$ $\Rightarrow$ $-(0.1,0.1)$; stays $\approx-(0.1,0.1)$ for constant $\g$.
\end{itemize}
Takeaway: GD/mom.\ move in proportion to the gradient (steep axis dominates); adaptive methods equalize the axes, so flat directions move as fast as steep ones.
